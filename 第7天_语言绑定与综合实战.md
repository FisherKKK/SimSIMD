## 第一部分：Python绑定深入剖析

### 1.1 为什么Python绑定很难做好？

**挑战**：
```python
# Python调用C函数的开销
def distance(a, b):
    return simsimd_cos_f32(a, b)  # C函数

开销分解：
1. 参数解析：~100ns（使用PyArg_ParseTuple）
2. 类型检查：~50ns
3. 数据转换：~20ns
4. 实际计算：~10ns（1536维向量）
5. 返回值打包：~30ns

总计：~210ns，其中计算只占 4.7%！
```

**常见方案的问题**：
```
❌ PyBind11：
   - 便利但慢（模板开销大）
   - 生成大量样板代码
   - 不支持Fast Calling Convention

❌ SWIG：
   - 代码生成复杂
   - 难以调试
   - 过时的设计

✅ Pure CPython API：
   - 完全控制
   - 最低开销
   - 支持Fast Calling Convention
   - 但需要手写大量代码
```

### 1.2 CPython Calling Conventions

**传统方式（METH_VARARGS | METH_KEYWORDS）**：

```c
// 函数签名
static PyObject* distance_slow(
    PyObject* self,
    PyObject* args,      // 元组（位置参数）
    PyObject* kwargs) {  // 字典（关键字参数）

    PyObject *a_obj, *b_obj;
    char* metric = "cos";

    // 动态解析（慢！）
    static char* kwlist[] = {"a", "b", "metric", NULL};
    if (!PyArg_ParseTupleAndKeywords(
        args, kwargs, "OO|s", kwlist,
        &a_obj, &b_obj, &metric))
        return NULL;

    // ... 处理逻辑
}

// 注册
{"distance", (PyCFunction)distance_slow, METH_VARARGS | METH_KEYWORDS, "..."}
```

**Fast Calling Convention（METH_FASTCALL）**：

```c
// 函数签名
static PyObject* distance_fast(
    PyObject* self,
    PyObject* const* args,     // C数组指针
    Py_ssize_t nargs,           // 位置参数数量
    PyObject* kwnames) {        // 关键字参数名元组

    // 直接访问，无需解析！
    if (nargs < 2) {
        PyErr_SetString(PyExc_TypeError, "需要至少2个参数");
        return NULL;
    }

    PyObject* a_obj = args[0];
    PyObject* b_obj = args[1];

    // 解析关键字参数
    char* metric = "cos";
    if (kwnames) {
        Py_ssize_t nkwargs = PyTuple_Size(kwnames);
        for (Py_ssize_t i = 0; i < nkwargs; ++i) {
            PyObject* key = PyTuple_GET_ITEM(kwnames, i);
            PyObject* value = args[nargs + i];

            if (PyUnicode_CompareWithASCIIString(key, "metric") == 0) {
                metric = PyUnicode_AsUTF8(value);
            }
        }
    }

    // ... 处理逻辑
}

// 注册
{"distance", (PyCFunction)distance_fast, METH_FASTCALL | METH_KEYWORDS, "..."}
```

**性能对比**：
```
PyArg_ParseTupleAndKeywords:  ~100ns
Fast Calling Convention:       ~10ns

加速：10x！
```

### 1.3 Buffer Protocol - 零拷贝数据访问

**问题**：如何高效访问NumPy/PyTorch/TensorFlow的数据？

```c
// 获取buffer信息
static int parse_tensor(PyObject* obj, Py_buffer* buffer, simsimd_datatype_t* datatype) {
    // 尝试获取buffer
    if (PyObject_GetBuffer(obj, buffer, PyBUF_STRIDES | PyBUF_FORMAT) == -1) {
        PyErr_SetString(PyExc_TypeError, "对象不支持buffer protocol");
        return -1;
    }

    // 检查维度
    if (buffer->ndim != 1) {
        PyBuffer_Release(buffer);
        PyErr_Format(PyExc_ValueError, "只支持1维数组，得到 %d 维", buffer->ndim);
        return -1;
    }

    // 解析数据类型
    if (buffer->format == NULL) {
        *datatype = simsimd_datatype_f32_k;  // 默认
    }
    else if (strcmp(buffer->format, "f") == 0) {
        *datatype = simsimd_datatype_f32_k;
    }
    else if (strcmp(buffer->format, "d") == 0) {
        *datatype = simsimd_datatype_f64_k;
    }
    else if (strcmp(buffer->format, "e") == 0) {
        *datatype = simsimd_datatype_f16_k;
    }
    // ... 更多类型
    else {
        PyBuffer_Release(buffer);
        PyErr_Format(PyExc_ValueError, "不支持的数据类型: %s", buffer->format);
        return -1;
    }

    return 0;
}
```

**核心函数示例**：

```c
static PyObject* api_cdist(
    PyObject* self,
    PyObject* const* args,
    Py_ssize_t nargs,
    PyObject* kwnames) {

    // 参数：a, b, metric, threads, dtype, out_dtype
    if (nargs < 2) {
        PyErr_SetString(PyExc_TypeError, "至少需要2个参数");
        return NULL;
    }

    PyObject* a_obj = args[0];
    PyObject* b_obj = args[1];

    // 默认值
    char const* metric_str = "cosine";
    size_t threads = 0;

    // 解析关键字参数
    if (kwnames) {
        Py_ssize_t nkwargs = PyTuple_Size(kwnames);
        for (Py_ssize_t i = 0; i < nkwargs; ++i) {
            PyObject* key = PyTuple_GET_ITEM(kwnames, i);
            PyObject* value = args[nargs + i];

            char const* key_str = PyUnicode_AsUTF8(key);
            if (strcmp(key_str, "metric") == 0) {
                metric_str = PyUnicode_AsUTF8(value);
            }
            else if (strcmp(key_str, "threads") == 0) {
                threads = PyLong_AsSize_t(value);
            }
        }
    }

    // 获取buffer
    Py_buffer a_buffer, b_buffer;
    simsimd_datatype_t dtype_a, dtype_b;

    if (parse_tensor(a_obj, &a_buffer, &dtype_a) == -1) return NULL;
    if (parse_tensor(b_obj, &b_buffer, &dtype_b) == -1) {
        PyBuffer_Release(&a_buffer);
        return NULL;
    }

    // 选择度量函数
    simsimd_metric_kind_t metric_kind = parse_metric(metric_str);

    // 分配输出矩阵
    Py_ssize_t a_count = a_buffer.len / a_buffer.itemsize;
    Py_ssize_t b_count = b_buffer.len / b_buffer.itemsize;

    npy_intp dims[2] = {a_count, b_count};
    PyArrayObject* result = (PyArrayObject*)PyArray_SimpleNew(2, dims, NPY_FLOAT64);

    // 并行计算距离矩阵
    #pragma omp parallel for collapse(2) schedule(dynamic) if(threads > 1)
    for (Py_ssize_t i = 0; i < a_count; ++i) {
        for (Py_ssize_t j = 0; j < b_count; ++j) {
            simsimd_distance_t distance;

            // 调用SIMD函数
            simsimd_metric_punned_t metric_func = simsimd_metric_punned(
                metric_kind, dtype_a, simsimd_cap_any_k
            );

            metric_func(
                (char*)a_buffer.buf + i * a_buffer.strides[0],
                (char*)b_buffer.buf + j * b_buffer.strides[0],
                a_buffer.len / a_buffer.itemsize,
                &distance
            );

            // 写入结果
            *(double*)PyArray_GETPTR2(result, i, j) = distance;
        }
    }

    // 释放buffer
    PyBuffer_Release(&a_buffer);
    PyBuffer_Release(&b_buffer);

    return (PyObject*)result;
}
```

### 3.1 Benchmarking方法论

**原则**：
```
1. 预热（Warm-up）：
   - 前几次运行用于预热CPU缓存
   - 跳过前N次测量

2. 多次测量：
   - 重复1000+次
   - 计算平均值、中位数、标准差

3. 隔离变量：
   - 固定输入大小
   - 固定线程数
   - 关闭动态频率调整

4. 对比基线：
   - NumPy / SciPy
   - 原生Python
   - 标量实现
```

### 3.2 Python Benchmark示例

```python
import simsimd
import numpy as np
import time
from scipy.spatial.distance import cosine as scipy_cosine

def benchmark_distance(n_vectors=1000, dimensions=1536, iterations=1000):
    """基准测试距离计算"""

    # 生成数据
    A = np.random.rand(n_vectors, dimensions).astype(np.float32)
    B = np.random.rand(n_vectors, dimensions).astype(np.float32)

    print(f"=== 基准测试: {n_vectors}个向量, {dimensions}维 ===\n")

    # 1. SimSIMD（单对）
    start = time.perf_counter()
    for _ in range(iterations):
        for i in range(n_vectors):
            simsimd.cosine(A[i], B[i])
    simsimd_time = time.perf_counter() - start
    print(f"SimSIMD (单对):    {simsimd_time:.3f}s")

    # 2. SimSIMD（批量）
    start = time.perf_counter()
    for _ in range(iterations):
        distances = simsimd.cdist(A, B, metric="cosine", threads=1)
    simsimd_batch_time = time.perf_counter() - start
    print(f"SimSIMD (批量):    {simsimd_batch_time:.3f}s")

    # 3. SciPy
    start = time.perf_counter()
    for _ in range(iterations):
        for i in range(n_vectors):
            scipy_cosine(A[i], B[i])
    scipy_time = time.perf_counter() - start
    print(f"SciPy:             {scipy_time:.3f}s")

    # 4. NumPy（手动实现）
    start = time.perf_counter()
    for _ in range(iterations):
        for i in range(n_vectors):
            a, b = A[i], B[i]
            dot = np.dot(a, b)
            norm_a = np.linalg.norm(a)
            norm_b = np.linalg.norm(b)
            cos_sim = dot / (norm_a * norm_b)
            cos_dist = 1 - cos_sim
    numpy_time = time.perf_counter() - start
    print(f"NumPy (手动):      {numpy_time:.3f}s")

    print(f"\n加速比（vs SciPy）:")
    print(f"  SimSIMD单对: {scipy_time / simsimd_time:.2f}x")
    print(f"  SimSIMD批量: {scipy_time / simsimd_batch_time:.2f}x")

    # 验证正确性
    simsimd_result = simsimd.cosine(A[0], B[0])
    scipy_result = scipy_cosine(A[0], B[0])
    print(f"\n正确性检查:")
    print(f"  SimSIMD: {simsimd_result:.6f}")
    print(f"  SciPy:   {scipy_result:.6f}")
    print(f"  误差:    {abs(simsimd_result - scipy_result):.2e}")

if __name__ == "__main__":
    benchmark_distance()
```

### 4.1 向量数据库（Vector Search）

**场景**：语义搜索引擎

```python
import simsimd
import numpy as np
from typing import List, Tuple

class VectorIndex:
    """简单的向量索引"""

    def __init__(self, embeddings: np.ndarray):
        """
        Args:
            embeddings: shape (n_vectors, dimensions)
        """
        self.embeddings = embeddings.astype(np.float32)
        self.n_vectors, self.dimensions = embeddings.shape

    def search(self, query: np.ndarray, top_k: int = 10) -> List[Tuple[int, float]]:
        """
        查找最相似的top_k个向量

        Args:
            query: shape (dimensions,)
            top_k: 返回的结果数量

        Returns:
            List of (index, distance) tuples
        """
        query = query.astype(np.float32)

        # 使用SimSIMD批量计算距离
        distances = np.array([
            simsimd.cosine(query, self.embeddings[i])
            for i in range(self.n_vectors)
        ])

        # 或者更快：使用cdist
        # distances = simsimd.cdist(
        #     query.reshape(1, -1),
        #     self.embeddings,
        #     metric="cosine"
        # )[0]

        # 找到top-k最小距离
        top_indices = np.argpartition(distances, top_k)[:top_k]
        top_indices = top_indices[np.argsort(distances[top_indices])]

        results = [(int(idx), float(distances[idx])) for idx in top_indices]
        return results

# 使用示例
# 1. 构建索引
embeddings = np.random.rand(100000, 1536).astype(np.float32)
index = VectorIndex(embeddings)

# 2. 搜索
query = np.random.rand(1536).astype(np.float32)
results = index.search(query, top_k=10)

print("Top 10 最相似的向量:")
for idx, distance in results:
    print(f"  索引 {idx}: 距离 {distance:.4f}")
```

### 4.2 推荐系统（Item Similarity）

```python
class ItemRecommender:
    """基于物品相似度的推荐"""

    def __init__(self, item_features: np.ndarray):
        """
        Args:
            item_features: shape (n_items, n_features)
        """
        self.item_features = item_features.astype(np.float32)
        self.n_items = item_features.shape[0]

        # 预计算物品之间的相似度矩阵
        print("预计算相似度矩阵...")
        self.similarity_matrix = self._compute_similarity_matrix()

    def _compute_similarity_matrix(self) -> np.ndarray:
        """计算物品相似度矩阵"""

        # 使用SimSIMD批量计算
        # 注意：cosine distance，需要转换为similarity
        distances = simsimd.cdist(
            self.item_features,
            self.item_features,
            metric="cosine",
            threads=8  # 并行计算
        )

        # 转换距离为相似度：similarity = 1 - distance
        similarities = 1.0 - distances

        return similarities

    def recommend(self, item_id: int, top_k: int = 5) -> List[Tuple[int, float]]:
        """推荐与给定物品最相似的物品"""

        similarities = self.similarity_matrix[item_id]

        # 排除自己
        similarities[item_id] = -1

        # 找到top-k
        top_indices = np.argpartition(similarities, -top_k)[-top_k:]
        top_indices = top_indices[np.argsort(-similarities[top_indices])]

        results = [(int(idx), float(similarities[idx])) for idx in top_indices]
        return results

# 使用示例
# 10000个物品，每个128维特征
item_features = np.random.rand(10000, 128).astype(np.float32)
recommender = ItemRecommender(item_features)

# 为物品ID=42推荐相似物品
recommendations = recommender.recommend(42, top_k=5)

print("推荐物品:")
for item_id, similarity in recommendations:
    print(f"  物品 {item_id}: 相似度 {similarity:.4f}")
```

### 4.3 异常检测（Anomaly Detection）

```python
class AnomalyDetector:
    """基于距离的异常检测"""

    def __init__(self, normal_data: np.ndarray):
        """
        Args:
            normal_data: shape (n_samples, n_features) - 正常数据
        """
        self.normal_data = normal_data.astype(np.float32)

        # 计算正常数据的中心
        self.center = np.mean(normal_data, axis=0).astype(np.float32)

        # 计算正常数据到中心的距离分布
        self.normal_distances = np.array([
            simsimd.l2sq(sample, self.center)
            for sample in self.normal_data
        ])

        # 设置阈值（例如：95分位数）
        self.threshold = np.percentile(self.normal_distances, 95)

        print(f"异常检测阈值: {self.threshold:.4f}")

    def is_anomaly(self, sample: np.ndarray) -> Tuple[bool, float]:
        """
        检测样本是否异常

        Returns:
            (is_anomaly, distance)
        """
        sample = sample.astype(np.float32)
        distance = simsimd.l2sq(sample, self.center)

        return distance > self.threshold, float(distance)

    def detect_batch(self, samples: np.ndarray) -> np.ndarray:
        """批量检测"""
        samples = samples.astype(np.float32)

        # 批量计算距离
        distances = simsimd.cdist(
            samples,
            self.center.reshape(1, -1),
            metric="l2sq"
        )[:, 0]

        return distances > self.threshold

# 使用示例
# 正常数据
normal_data = np.random.randn(1000, 128).astype(np.float32)
detector = AnomalyDetector(normal_data)

# 测试数据（包含异常）
test_data = np.random.randn(100, 128).astype(np.float32)
test_data[50] *= 5  # 人工注入异常

# 批量检测
anomalies = detector.detect_batch(test_data)
print(f"检测到 {anomalies.sum()} 个异常样本")
```

### 5.1 项目：语义相似度搜索引擎

**需求**：

**完整实现**：

```python
import simsimd
import numpy as np
from typing import List, Dict, Any, Optional
import json

class SemanticSearchEngine:
    """语义搜索引擎"""

    def __init__(self):
        self.embeddings = None
        self.metadata = []
        self.index = {}

    def add_documents(self, docs: List[Dict[str, Any]], embeddings: np.ndarray):
        """
        添加文档到索引

        Args:
            docs: 文档元数据列表
            embeddings: 文档embedding矩阵 (n_docs, dimensions)
        """
        if self.embeddings is None:
            self.embeddings = embeddings.astype(np.float32)
        else:
            self.embeddings = np.vstack([self.embeddings, embeddings.astype(np.float32)])

        start_idx = len(self.metadata)
        self.metadata.extend(docs)

        # 构建倒排索引（用于过滤）
        for i, doc in enumerate(docs, start=start_idx):
            for key, value in doc.items():
                if key not in self.index:
                    self.index[key] = {}
                if value not in self.index[key]:
                    self.index[key][value] = []
                self.index[key][value].append(i)

    def search(
        self,
        query_embedding: np.ndarray,
        top_k: int = 10,
        filters: Optional[Dict[str, Any]] = None,
        metric: str = "cosine"
    ) -> List[Dict[str, Any]]:
        """
        搜索相似文档

        Args:
            query_embedding: 查询向量
            top_k: 返回结果数
            filters: 过滤条件 {"key": "value"}
            metric: 距离度量（"cosine", "l2sq", "dot"）
        """
        query_embedding = query_embedding.astype(np.float32)

        # 应用过滤器
        if filters:
            candidate_indices = self._apply_filters(filters)
            candidate_embeddings = self.embeddings[candidate_indices]
        else:
            candidate_indices = list(range(len(self.metadata)))
            candidate_embeddings = self.embeddings

        # 批量计算距离
        if metric == "cosine":
            distances = simsimd.cdist(
                query_embedding.reshape(1, -1),
                candidate_embeddings,
                metric="cosine"
            )[0]
        elif metric == "l2sq":
            distances = simsimd.cdist(
                query_embedding.reshape(1, -1),
                candidate_embeddings,
                metric="l2sq"
            )[0]
        elif metric == "dot":
            # 点积：越大越相似，需要取负
            distances = -simsimd.cdist(
                query_embedding.reshape(1, -1),
                candidate_embeddings,
                metric="dot"
            )[0]
        else:
            raise ValueError(f"不支持的度量: {metric}")

        # Top-K
        top_indices = np.argpartition(distances, min(top_k, len(distances) - 1))[:top_k]
        top_indices = top_indices[np.argsort(distances[top_indices])]

        # 构建结果
        results = []
        for idx in top_indices:
            actual_idx = candidate_indices[idx]
            results.append({
                "id": actual_idx,
                "metadata": self.metadata[actual_idx],
                "distance": float(distances[idx])
            })

        return results

    def _apply_filters(self, filters: Dict[str, Any]) -> List[int]:
        """应用过滤条件"""
        candidate_sets = []

        for key, value in filters.items():
            if key in self.index and value in self.index[key]:
                candidate_sets.append(set(self.index[key][value]))

        # 取交集
        if candidate_sets:
            candidates = set.intersection(*candidate_sets)
            return list(candidates)
        else:
            return []

    def save(self, filepath: str):
        """保存索引"""
        np.savez_compressed(
            filepath,
            embeddings=self.embeddings,
            metadata=json.dumps(self.metadata),
            index=json.dumps(self.index)
        )

    @classmethod
    def load(cls, filepath: str):
        """加载索引"""
        data = np.load(filepath, allow_pickle=True)
        engine = cls()
        engine.embeddings = data['embeddings']
        engine.metadata = json.loads(str(data['metadata']))
        engine.index = json.loads(str(data['index']))
        return engine

# 使用示例
def demo_search_engine():
    # 1. 创建搜索引擎
    engine = SemanticSearchEngine()

    # 2. 添加文档
    docs = [
        {"title": "Python教程", "category": "编程", "author": "Alice"},
        {"title": "机器学习入门", "category": "AI", "author": "Bob"},
        {"title": "深度学习实战", "category": "AI", "author": "Charlie"},
        {"title": "Web开发指南", "category": "编程", "author": "Alice"},
    ]

    embeddings = np.random.rand(4, 128).astype(np.float32)
    engine.add_documents(docs, embeddings)

    # 3. 搜索
    query = np.random.rand(128).astype(np.float32)

    # 无过滤
    results = engine.search(query, top_k=2, metric="cosine")
    print("=== 搜索结果（无过滤） ===")
    for r in results:
        print(f"  {r['metadata']['title']}: {r['distance']:.4f}")

    # 带过滤
    results = engine.search(
        query,
        top_k=2,
        filters={"category": "AI"},
        metric="cosine"
    )
    print("\n=== 搜索结果（过滤category=AI） ===")
    for r in results:
        print(f"  {r['metadata']['title']}: {r['distance']:.4f}")

    # 4. 保存和加载
    engine.save("search_index.npz")
    loaded_engine = SemanticSearchEngine.load("search_index.npz")
    print("\n索引已保存并重新加载")

if __name__ == "__main__":
    demo_search_engine()
```
